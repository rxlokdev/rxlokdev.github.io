// throwaway.inj — core injection via LoadLibrary
#include <Windows.h>
#include <iostream>

bool Inject(DWORD pid, const char* dllPath) {
  HANDLE hProc = OpenProcess(PROCESS_ALL_ACCESS, FALSE, pid);
  if (!hProc) return false;
  LPVOID mem = VirtualAllocEx(hProc, nullptr, strlen(dllPath)+1, MEM_COMMIT, PAGE_READWRITE);
  WriteProcessMemory(hProc, mem, dllPath, strlen(dllPath)+1, nullptr);
  CreateRemoteThread(hProc, nullptr, 0,
    (LPTHREAD_START_ROUTINE)GetProcAddress(GetModuleHandle("kernel32"), "LoadLibraryA"),
    mem, 0, nullptr);
  CloseHandle(hProc);
  return true;
}

Usage

1

Clone the repository

git clone https://github.com/rxlokdev/throwaway.inj

2

Build with MSVC or MinGW

Open in Visual Studio or compile with g++ main.cpp -o throwaway -lkernel32

3

Select your target process

Browse the process list, click to select your target — PID and process name auto-fill in the Target panel.

4

Browse for your DLL and inject

Click Browse to pick your .dll file, then hit Inject DLL. Check the log panel for status.

Requirements

  • Windows 10 or later (x64 only)
  • MSVC (Visual Studio 2019+) or MinGW-w64
  • DirectX 11 capable graphics card
  • Target process must be x64
